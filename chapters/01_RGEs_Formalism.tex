\section{\texorpdfstring{$S$-Matrix and Dilatation Operator}{S-Matrix and Dilatation Operator}}

In this section, we will closely follow Ref.~\cite{EliasMiro:2020tdv} to derive a central relation satisfied by the scattering matrix, $S$, and the dilatation operator, $D$, due to Caron-Huot and Wilhelm \cite{Caron-Huot:2016cwu}, which will allow us to extract the anomalous dimensions of operators via unitarity cuts.

The fundamental objects we deal with are the \textit{form factors} of local and gauge-invariant operators $\mathcal{O}_i$ entering the effective Lagrangian
\begin{equation}\label{eq:RGELagrangian}
    \mathcal L_{\rm EFT} = \sum_i \frac{c_i}{\Lambda^{[\mathcal O_i]-4}}\,\mathcal O_i\,,
\end{equation}
which are defined as the matrix elements
\begin{equation}\label{eq:RGEformfactor}
    F_i(\vec n;q) = \frac{1}{\Lambda^{[\mathcal O_i]-4}}\; {}_{\text{out}}\!\mel{\vec n}{\mathcal O_i(q)}{0}\,,
\end{equation}
taken between the vacuum and an outgoing asymptotic multi-particle state ${}_{\text{out}}\!\bra{\vec n}$,
where $q$ is the off-shell momentum injected by the operator.
Therefore, a form factor is an object that sits between a Green's function and a scattering amplitude: its external legs are on shell, $p_1^2 = \dotsb = p_n^2 = 0$ in the massless limit, as in amplitudes, but one local insertion is kept off shell, $q^2 \neq 0$. The information about the renormalization of $\mathcal O_i$ is retained in this insertion.

By Lorentz invariance, form factors depend on the four-momenta of the outgoing particles through the Mandelstam invariants $s_{ab} = 2 p_a\cdot p_b$, i.e., $F_i = F_i(\{s_{ab}+i\epsilon\})$, where we have introduced $\epsilon>0$ due to the Feynman $i\epsilon$ prescription.
There are two ways to relate $F_i(\{s_{ab}+i\epsilon\})$ to some function evaluated on the complex-conjugated domain $\{s_{ab}-i\epsilon\}$.
\begin{itemize}
    \item The first one exploits the analyticity property of form factors: the complex conjugation of form factors amounts to transforming time-ordered propagators, with denominators of the form $s_{ab}+i\epsilon$, into anti-time-ordered propagators, with denominators of the form $s_{ab}-i\epsilon$.
    Thus, under complex conjugation, $\{s_{ab}+i\epsilon\}$ are effectively replaced by $\{s_{ab}-i\epsilon\}$, and the analyticity relation reads
    \begin{equation}\label{eq:partial1}
        F_i(\{s_{ab}+i\epsilon\}) = F^*_i(\{s_{ab}-i\epsilon\})\,.
    \end{equation}

    \item The second one consists of analytically continuing all the momenta $p_a$ to the complex plane and rotating them along a counter-clockwise circle with a common phase $\alpha$:
    \begin{equation}
        p_a \longrightarrow e^{i\alpha}p_a\,.
    \end{equation}
    Correspondingly, the Mandelstam invariants are rotated by $2 \alpha$ and the
    form factors are transformed as
    \begin{equation}
        F_i \longrightarrow e^{i\alpha D} F_i \,,
    \end{equation}
    where $D$ is the dilatation operator, which is the generator of the transformation and is defined as
    \begin{equation}
        D = \sum_{a\in\vec n} p_a \cdot \frac{\partial}{\partial p_a}\,.
    \end{equation}
    For $\alpha=\pi$, as illustrated in Figure~\ref{fig:alphapi}, the Mandelstam invariants are back to their original values, but with the opposite imaginary part. Therefore,
    \begin{equation}\label{eq:partial2}
        F_i(\{s_{ab}-i\epsilon\}) = e^{i\pi D}  F_i(\{s_{ab}+i\epsilon\})\,.
    \end{equation}
\end{itemize}

Combining Eqs.~\eqref{eq:partial1} and \eqref{eq:partial2} we obtain
\begin{equation}\label{eq:partial4}
    F_i(\{s_{ab}+i\epsilon\})=e^{-i\pi D} F_i^*(\{s_{ab}+i\epsilon\}) \,.
\end{equation}

Next, we can exploit unitarity in the form of the completeness relation satisfied by the multi-particle asymptotic states $\ket{\vec n}$, which form a complete basis of the Hilbert space:
\begin{equation}
    \mathbb 1 = \sum_{\vec n} \int \text{dLIPS}_{\vec n}\, \ket{\vec n}\bra{\vec n}\,,
\end{equation}
where
\begin{equation}
    \text{dLIPS}_{\vec n} = \frac{1}{S_{\vec n}} \prod_{k\in \vec n}\frac{\dd^3 p_k}{(2\pi)^3}\frac{1}{2E_k}
\end{equation}
is the Lorentz-invariant phase-space measure associated with the state $\ket{\vec n}$ and $S_{\vec n}$ is its symmetry factor (for instance, $S_{\vec n} = n!$ if $\ket{\vec n}$ is composed of $n$ identical particles).
We can exploit the completeness relation to rewrite a generic form factor $F_i$ as
\begin{equation}\label{eq:partial3}
    F_i(\vec n;q) = \frac{1}{\Lambda^{[\mathcal O_i]-4}} \sum_{\vec m} \int \text{dLIPS}_{\vec m} \;{}_{\text{out}}\!\braket*{\vec n}{\vec m}\!{}_{\text{in}}\; {}_{\text{in}}\!\mel*{\vec m}{\mathcal O_i(q)}{0}\,.
\end{equation}
The overlap ${}_{\text{out}}\!\braket*{\vec n}{\vec m}\!{}_{\text{in}}$ is, by definition, an element
of the scattering matrix, ${}_{\text{out}}\!\braket*{\vec n}{\vec m}\!{}_{\text{in}} = S(\vec n ; \vec m)$.

The final ingredient is provided by the CPT theorem. 
Being represented in the Hilbert space by an anti-linear and anti-unitary operator $\Theta$, CPT
interchanges incoming and outgoing states inside an inner product, while acting on local operators in position space as
$\Theta\,\mathcal{O}_i(x)\,\Theta^{-1} = \mathcal{O}_i^\dagger(-x)$, or in momentum space as
\begin{equation}
    \Theta\,\mathcal O_i(q)\,\Theta^{-1} = \mathcal O_i^\dagger(q)\,.
\end{equation}
Inserting $\Theta^{-1}\Theta$ twice into ${}_{\text{out}}\!\mel*{\vec n}{\mathcal O_i(q)}{0}$
therefore yields
\begin{equation}\label{eq:RGECPT}
    {}_{\text{out}}\!\mel*{\vec n}{\mathcal O_i(q)}{0} = \mel*{0}{\mathcal O_i^\dagger(q)}{\vec n}\!{}_{\text{in}}\,,
\end{equation}
which implies that, for a Hermitian operator $\mathcal O_i^\dagger = \mathcal O_i$,
\begin{equation}
    {}_{\text{in}}\!\mel*{\vec m}{\mathcal O_i(q)}{0} = \mel*{0}{\mathcal O_i(q)}{\vec m}\!{}_{\text{out}} = \left({}_{\text{out}}\!\mel*{\vec m}{\mathcal O_i(q)}{0}\right)^* = \frac{1}{\Lambda^{4-[\mathcal O_i]}} \, F_i^*(\vec m;q)\,.
\end{equation}

Thus, we can rewrite Eq.~\eqref{eq:partial3} as
\begin{equation}
    F_i(\vec n;q) = \sum_{\vec m}\int \text{dLIPS}_{\vec m} \, S(\vec n;\vec m) F_i^*(\vec m;q)\,,
\end{equation}
or, more concisely, as $F_i = SF_i^*$. Finally, combining it with Eq.~\eqref{eq:partial4} we obtain the non-perturbative relation \cite{Caron-Huot:2016cwu}
\begin{equation}\label{eq:CHWmaster}
    e^{-i\pi D} F_i^* = S F_i^*\,.
\end{equation}
Its interpretation consists in identifying the dilatation operator $D$ as minus the phase of the $S$-matrix, divided by $\pi$.

\section{General Operator Mixing}

In perturbation theory, once loop integrals are performed, both ultraviolet and infrared divergences appear, and one needs to introduce a regulator.
Dimensional regularization \cite{Bollini:1972ui,tHooft:1972tcz,Ashmore:1972uj} introduces the arbitrary scale $\mu$, upon which no physical quantity may ultimately depend.
For the form factors defined in Eq.~\eqref{eq:RGEformfactor}, the statement about the independence on the 't~Hooft scale $\mu$ takes the form of the Callan--Symanzik equation \cite{Callan:1970yg,Symanzik:1970rt}
    \begin{equation}
    \label{eq:CSEq}
    \left( \mu\frac{\partial}{\partial \mu} +\sum_g\beta_g \frac{\partial}{\partial g} + \sum_i \beta_i \frac{\partial}{\partial c_i}- \gamma_{\text{IR}}\right)F_j + \sum_i \frac{\partial \beta_i}{\partial c_j} F_i =   0\,,
\end{equation}
which is valid in the massless limit. In this equation, $g$ collectively denotes the couplings of the renormalizable theory, and we have introduced the beta functions of the renormalizable couplings and of the Wilson coefficients,
\begin{equation}\label{eq:betadefs}
    \beta_g = \mu\dv[]{g}{\mu}\,, \qquad\qquad \beta_i = \mu\dv[]{c_i}{\mu}\,,
\end{equation}
while $\gamma_{\text{IR}} = \gamma_{\text{IR}}(\vec n)$ is the infrared anomalous dimension of the external state $\vec n$, which originates from soft radiation exchanged between any pair of legs and from collinear radiation emitted along a single leg.
Such radiation is universal: having a wavelength much larger than the inverse of the hard scales $\sqrt{\abs*{s_{ab}}}$ set by the external kinematics, it cannot resolve the local operator that created the state, and the associated divergences factorize from the hard process into a factor determined solely by the momenta and the quantum numbers of the particles in $\vec n$ \cite{Sterman:2002qn,Becher:2009cu}.
Thus, $\gamma_{\text{IR}}$ generates no
operator mixing and is proportional to the identity in operator space.

In the neighborhood of the Gaussian fixed point $(*)$ --- where $c_i = 0$ for all $i$ --- the beta functions of the Wilson coefficients can be Taylor expanded as
\begin{equation}\label{eq:Taylorbeta}
    \beta_i = \sum_{k>0}\frac{1}{k!}\sum_{j_1,\dotsc,j_k}\gamma_{i\leftarrow j_1,\dotsc,j_k}\,c_{j_1}\dotsm c_{j_k}
    = \sum_j  \gamma_{i\leftarrow j}\,c_j + \frac{1}{2}\sum_{j_1,j_2}\gamma_{i\leftarrow j_1,j_2}\,c_{j_1}c_{j_2} + \dots \,,
\end{equation}
which defines the \textit{anomalous-dimension tensors} for an arbitrary number of operator insertions,
\begin{equation}\label{eq:gammagen}
    \gamma_{i\leftarrow j_1,\dotsc,j_k} = \left.\frac{\partial^k \beta_i}{\partial c_{j_1}\dotsm \partial c_{j_k}}\right|_{*}\,,
\end{equation}
each of which admits a perturbative expansion in the couplings of the leading-order Lagrangian, $\gamma_{i\leftarrow j_1,\dotsc,j_k} = \sum_{L\ge 1}\gamma^{(L)}_{i\leftarrow j_1,\dotsc,j_k}$.
For $k=1$ this reduces to the anomalous-dimension matrix, $\gamma_{i\leftarrow j} = \partial\beta_i/\partial c_j|_*$, governing the mixing of operators of the same dimension, while $k\ge 2$ describes the mixing of $k$ lower-dimensional operators into a higher-dimensional one.

In the following, we show how the Callan--Symanzik equation \eqref{eq:CSEq} can be combined with Eq.~\eqref{eq:CHWmaster} in order to extract one-loop and two-loop anomalous dimensions via unitarity cuts of form factors, that is, by isolating their discontinuities.

\subsection{One-Loop Mixing}

The Callan--Symanzik equation \eqref{eq:CSEq}
can be expanded at one loop as
\begin{equation}
    \label{eq:CSoneloop}
    \mu\frac{\partial}{\partial \mu} F_j^{(1)} = -\left(\sum_g\beta_g^{(1)} \frac{\partial}{\partial g} + \sum_i \beta_i^{(1)} \frac{\partial}{\partial c_i}- \gamma_{\text{IR}}^{(1)}\right)F_j^{(0)} - \sum_i \frac{\partial \beta_i^{(1)}}{\partial c_j} F_i^{(0)}
\end{equation}
since the perturbative expansions of $\beta_g$, $\beta_i$, and $\gamma_{\text{IR}}$ all start at one loop; therefore, they can act only on the tree-level form factors.
On the other hand, 
\begin{equation}\label{eq:DFoneloop}
    (D F_j)^{(1)} = -\mu\frac{\partial}{\partial \mu} F_i^{(1)} = -\frac{1}{\pi}(\mathcal M F_i)^{(1)}\,,
\end{equation}
where the first equality follows from the fact that, in dimensional regularization and in the massless limit, a form factor can depend on the kinematics solely through the dimensionless ratios $p_a\cdot p_b/\mu^2$, up to its overall engineering dimension, while the second equality follows from Eq.~\eqref{eq:CHWmaster} expanded to first order, with $S=\mathbb{1}+i\mathcal M$.
Equating Eqs.~\eqref{eq:CSoneloop} and \eqref{eq:DFoneloop}, we obtain
\begin{equation}\label{eq:oneloopgeneral}
    \left(\sum_g\beta_g^{(1)} \frac{\partial}{\partial g} + \sum_i \beta_i^{(1)} \frac{\partial}{\partial c_i}- \gamma_{\text{IR}}^{(1)}\right)F_j^{(0)} + \sum_i \frac{\partial \beta_i^{(1)}}{\partial c_j} F_i^{(0)} = -\frac{1}{\pi}(\mathcal M F_j)^{(1)}\,.
\end{equation}
The convolution on the right-hand side is a sum over all one-loop two-particle unitarity cuts
\begin{align}\label{eq:oneloopcut}
    (\mathcal M F_j)^{(1)}(\vec n) =
    &\sum_{m=2}^{n}\sum_{\{\ell_1,\ell_2\}}\int \text{dLIPS}_2
    \sum_{h_1,h_2} \mathcal M^{(0)}(1,\dotsc,m;\ell_1^{h_1},\ell_2^{h_2}) F_j^{(0)}(\ell_1^{h_1},\ell_2^{h_2},m+1,\dotsc,n)\nonumber \\&+\text{permutations}\,,
\end{align}
where $\mathcal M(\vec n;\vec m) = \mel{\vec n}{\mathcal M}{\vec m}$ and $\text{dLIPS}_n$ is the $n$-particle Lorentz-invariant phase-space measure.
Eq.~\eqref{eq:oneloopgeneral} holds for every external-state configuration.

We can considerably simplify Eq.~\eqref{eq:oneloopgeneral} 
by choosing the minimal configuration $\vec n_i$ created by the operator $\mathcal O_i$ as an external state. In this way, we can project the above equation onto
\begin{equation}
    \left(\frac{\partial \beta_i^{(1)}}{\partial c_j}  - \delta_{ij}\gamma_{i,\text{IR}}^{(1)}\right) F_i^{(0)} = -\frac{1}{\pi}(\mathcal M F_j)^{(1)}\,,
\end{equation}
%where the label $i$ has been used in the infrared anomalous dimension as a shorthand for the external state on which the form factor is evaluated.
where we used the shorthand notation $\gamma_{i,\text{IR}} = \gamma_{\text{IR}}(\vec n_i)$,
from which the one-loop master formulae follow \cite{Caron-Huot:2016cwu,EliasMiro:2020tdv}:
\begin{equation}
    \left(\gamma_{i\leftarrow j}^{(1)} - \delta_{ij}\gamma_{i,\text{IR}}^{(1)}\right) F_i|_*^{(0)} = -\frac{1}{\pi}(\mathcal M F_j)|_*^{(1)}
\end{equation}
for the case of linear operator mixing, and
\begin{equation}
    \gamma_{i\leftarrow j_1,\dotsc,j_k}^{(1)} \, F_i|_*^{(0)} = -\frac{1}{\pi}\left.\frac{\partial^{k-1}}{\partial c_{j_2} \dotsm \partial c_{j_k}}\right|_* (\mathcal M F_{j_1})^{(1)}
\end{equation}
for the case of $k$ higher-dimensional operator insertions, with $k\ge 2$.

\subsection{Two-Loop Mixing}

The two-loop master formula can be derived as follows. 
Expanding the exponential $e^{-i\pi D} - \mathbb{1} = -i\pi D - \frac{\pi^2}{2}D^2 + O(D^3)$ and dividing by $-i\pi$, Eq.~\eqref{eq:CHWmaster} becomes
\begin{equation}\label{eq:CHWexpanded}
    D F_j - \frac{i\pi}{2}D^2 F_j = -\frac{1}{\pi}\mathcal M F_j\,.
\end{equation}
Defining the operator $\Gamma$ as
\begin{equation}\label{eq:Gammadef}
    \Gamma F_j = \sum_i \frac{\partial \beta_i}{\partial c_j}F_i 
    + \sum_g\beta_g\frac{\partial F_j}{\partial g} + \sum_i \beta_i\frac{\partial F_j}{\partial c_i}- \gamma_{\text{IR}}F_j\,,
\end{equation}
which can be expanded as $\Gamma = \Gamma^{(1)} + \Gamma^{(2)}+\dots$,
the Callan--Symanzik equation \eqref{eq:CSEq} can be compactly written as
\begin{equation}\label{eq:DGamma}
    D F_j = -\mu\frac{\partial}{\partial\mu}F_j = \Gamma F_j\,.
\end{equation}

Inserting Eq.~\eqref{eq:DGamma} into Eq.~\eqref{eq:CHWexpanded} and collecting the terms of a given
order, the first order reproduces the one-loop master formula, while the second gives
\begin{equation}\label{eq:twoloopmaster}
    \Gamma^{(1)}F^{(1)}_j + \Gamma^{(2)}F^{(0)}_j
    - \frac{i\pi}{2}\Gamma^{(1)}\Gamma^{(1)}F^{(0)}_j = -\frac{1}{\pi}(\mathcal M F_j)^{(2)}\,,
\end{equation}
where $(\mathcal M F_j)^{(2)}$ collects the two-loop unitarity cuts, i.e., 
\begin{itemize}
    \item[(a)] the two-particle cuts of one-loop renormalized amplitudes against tree-level form factors,
    \item[(b)] the
two-particle cuts of tree-level amplitudes against one-loop renormalized form factors, and
\item[(c)] the three-particle
cuts of two tree-level amplitudes and form factors.
\end{itemize}
For more details, see Chapter~\ref{ch:2looptheorems} and Figure~\ref{fig:cuts}.

Following Ref.~\cite{Bern:2020ikv}, this equation can be considerably simplified by separating the real and imaginary parts.
Both $\Gamma^{(1)}$ and $F_j^{(0)}$ are real, so that the third term of Eq.~\eqref{eq:twoloopmaster} is purely imaginary and drops out of the real part. Thus, we are left with
\begin{equation}\label{eq:aaaa}
    \Gamma^{(1)}\Re F^{(1)}_j + \Gamma^{(2)}F^{(0)}_j
     = -\frac{1}{\pi}\Re (\mathcal M F_j)^{(2)}\,.
\end{equation}
By contrast, the imaginary part of Eq.~\eqref{eq:twoloopmaster} does not contain the two-loop anomalous dimension, and its right-hand side, $\Im(\mathcal M F_j)^{(2)}$, is nothing but the iterated two-particle unitarity cut $(\mathcal M \mathcal M F_j)^{(2)}$ due to the optical theorem.
This allows us to write $\Re (\mathcal M F_j)^{(2)}$ as
\begin{align}\label{eq:ReMReF}
    \Re (\mathcal M F_j)^{(2)}&=(\mathcal M F_j - \mathcal M\mathcal M F_j)^{(2)}
    = \bigg(\bigg(\mathcal M - \frac{1}{2}\mathcal M\mathcal M\bigg)\bigg(F_j - \frac{1}{2} \mathcal M F_j\bigg)\bigg)^{(2)}\nonumber \\&
    = (\Re \mathcal M  \Re F_j)^{(2)}\,,
\end{align}
so that Eq.~\eqref{eq:aaaa} becomes
\begin{equation}\label{eq:twoloopmaster2}
    \Gamma^{(1)}\Re F^{(1)}_j + \Gamma^{(2)}F^{(0)}_j
     = -\frac{1}{\pi} (\Re\mathcal M \Re F_j)^{(2)}\,.
\end{equation}

As at one loop, the equation simplifies further if the external state is chosen to be the minimal
configuration created by $\mathcal O_i$.
The tree-level form factor $F^{(0)}$ is then local and
therefore independent of both the marginal couplings and the Wilson coefficients, while the one-loop form factor $F^{(1)}$ is not local in general, and
the corresponding terms in $\Gamma^{(1)}\Re F^{(1)}$ have to be retained.
Projecting
Eq.~\eqref{eq:twoloopmaster2} as before, we obtain the two-loop master formula for linear
operator mixing \cite{Bern:2020ikv}
\begin{equation}\label{eq:twoloopmasterlinear}
    \left(\gamma_{i\leftarrow j}^{(2)} - \delta_{ij}\gamma_{i,\text{IR}}^{(2)}\right)F_i|_*^{(0)}
    = -\left.\left(\frac{1}{\pi}(\Re \mathcal M\Re F_j)^{(2)}+\Gamma^{(1)}\Re F^{(1)}_j \right)\right|_{*}
\end{equation}
and
\begin{equation}
    \gamma_{i\leftarrow j_1,\dotsc,j_k}^{(2)}\, F_i|_*^{(0)}
    = -\left.\frac{\partial^{k-1}}{\partial c_{j_2} \dotsm \partial c_{j_k}}\right|_* \left(\frac{1}{\pi}(\Re \mathcal M\Re F_{j_1})^{(2)}+\Gamma^{(1)}\Re F^{(1)}_{j_1} \right)
\end{equation}
for the case of $k$ higher-dimensional operator insertions, with $k\ge 2$.

\section{Leading Mass Effects}

As a natural consequence of operator mixing, chirality-violating and chirality-preserving operators generally mix under the renormalization flow.
In the case of EFTs defined below the electroweak scale, the required chirality flip proceeds through a fermion mass insertion. However, such a mass dependence cannot be directly implemented in the massless method of Ref.~\cite{Caron-Huot:2016cwu}. 

In order to circumvent this issue, we rely on the Higgs low-energy theorem~\cite{Ellis:1975ap,Shifman:1979eb}, which was originally introduced to estimate the properties of a light Higgs boson in analogy to how soft-pion theorems are used to study low-energy pion interactions~\cite{Adler:1964um,Cheung:2021yog,Balkin:2021dko,Bertuzzo:2023slg}. 
%
The key observation is that the fermionic Higgs interactions in the SM can be written in the following form:
%
\begin{align}
\mathcal{L}^{\textit{int}}_{H} = - \left( 1 + \frac{h}{v}\right) 
\sum_f m_f \bar f f \,,
\end{align}
%
where the sum is over all fermions in the theory. 
In the limit where the Higgs field $h$ has a vanishing 
four-momentum, $p_h\to 0$, $h$ becomes a constant field 
and its effect is equivalent to redefining all mass 
parameters as $m_f \to m_f \,(1 + h/v)$.
This immediately implies the following low-energy theorem~\cite{Ellis:1975ap,Shifman:1979eb}
%
\begin{align}
\lim_{\{p_{h}\} \to 0}\mathcal{M}(A \to B 
+ Nh) = 
\left(\sum_f \frac{m_f}{v}\frac{\partial}{\partial m_f}\right)^{\!N} \mathcal{M}(A\to B)\,,
\end{align}
%
relating the amplitudes of two processes differing by $N$ insertions of zero-momentum Higgs bosons. 
%
%The key idea consists in promoting fermionic masses to dynamical massless spurionic fields, whose vacuum expectation value is eventually set to the mass of the fermions. This is reminiscent of what happens in the Higgs mechanism, where fermion masses and Yukawa interactions are one-to-one related via the term $m_f (1 + h/v) \bar{f}f$. 
%
In practice, whenever an amplitude requires $N$ fermion mass insertions not to vanish, we consider an equivalent amplitude entailing $N$ extra massless Higgs fields; see Figure~\ref{fig:HMs}.

The number $N$ can be determined as follows. By dimensional 
analysis, we argue that whenever the anomalous dimension 
$\gamma_{i\leftarrow j_1, \dotsc, j_n}$ vanishes in the 
limit of massless fermions, possible mass effects must 
be of order $(m_f/\Lambda)^N$, where
%
\begin{equation}
\label{eq:Sup_Div_Deg}
N = 4-[\mathcal{O}_{i}] +\sum^n_{k=1} ([\mathcal{O}_{j_k}]-4)
\end{equation}
% 
is the superficial degree of divergence associated with the loop diagram under consideration (see Appendix~\ref{app:SDOD_EFT} for a derivation).
Notice that the number of needed Higgs insertions coincides with the superficial degree of divergence in Eq.~\eqref{eq:Sup_Div_Deg} because scaleless integrals vanish in dimensional regularization.
%
For $N < 0$, $\gamma_{i\leftarrow j_1, \dotsc, j_n}$ is trivially zero.
For $N \geq 0$, the anomalous dimension
$\gamma_{i\leftarrow j_1, \dotsc, j_n}$ 
is obtained by renormalizing the operator $\mathcal{O}^{Nh}_i= (h/v)^N\mathcal{O}_i/N!$
instead of $\mathcal{O}_i$.
%

\section{Conclusions}

In this chapter, we have generalized the application of on-shell and unitarity-based methods for evaluating renormalization group coefficients to account for the mixing of operators with different dimensions and leading mass effects, both of which play a fundamental role in the renormalization program of several effective field theories. 
%
In particular, we have derived master formulae accounting for the most general operator 
mixings up to two-loop order and shown how to include leading mass effects,
relying on the Higgs low-energy theorem. %These findings have been validated by reproducing well-established results in the literature, corresponding to popular effective field theories.

Applications of these findings are deferred to the two chapters that follow, where they are exploited to compute the one-loop ultraviolet anomalous dimensions of the axion-like particle effective field
theory, in Chapter~\ref{ch:ALP_RGEs}, and of a general effective gauge theory, in
Chapter~\ref{ch:generalEFT}.

The reach of the results derived here extends well beyond those examples.
They can be applied to a number of new-physics scenarios, 
defined above the TeV scale, in order to analyze their impact on 
low-energy observables (such as flavor-violating processes, 
electric and magnetic dipole moments, etc.)\ occurring at or 
below the GeV scale.
Such a large separation of scales demands the inclusion of 
running effects at two-loop order to obtain sensible predictions. 
While this is a very challenging task when approached with standard techniques, 
on-shell and unitarity-based methods offer a simpler, more efficient, 
and elegant way to reach this goal.

\section{Effective Field Theories and Superficial Degree of Divergence\label{app:SDOD_EFT}}

An $n$-particle amplitude related to the operator $\Ocal_i$ 
and entailing $m$ interaction vertices with the operators $\Ocal_{j_1},\dotsc,\Ocal_{j_m}$ can be schematically written as
\begin{equation}
    \mathcal M_{n} = C_{j_1}\dotsm C_{j_m} 
     \int \dd^4 k_1 \dotsm \dd^4 k_L \,\frac{\mathcal N
     (\{k\},\{p\})}{\mathcal D(\{k\},\{p\})} \,
    \alpha_1 \dotsm \alpha_{n} \,,
\end{equation}
where $\{p\}$ are external momenta and:
\begin{itemize}
    \item $C_{j_1},\dotsc, C_{j_m}$ are the dimensionful Wilson coefficients corresponding to the operators $\Ocal_{j_1},\dotsc,\Ocal_{j_m}$ 
    such that
    \begin{equation}
        [C_{j_k}]= 4 - [\Ocal_{j_k}]\,.
    \end{equation}        

    \item The $L$-loop integral has mass dimension 
    \begin{equation}
    \left[\int \dd^4 k_1\dotsi \dd^4 k_L\, \frac{\mathcal N(\{k\},\{p\})}{\mathcal D(\{k\},\{p\})}\right]=
    \mathsf D + n_\partial\,,
    \end{equation}        
    where $\mathsf D$ is the degree of divergence of 
    $\mathcal M_{n}$ and $n_\partial$ is the number 
    of derivatives contained in $\Ocal_i$.

    \item Depending on the bosonic/fermionic nature of the particles involved, $\alpha_1,\dotsc, \alpha_{n} \in \{1,\epsilon_\mu,\epsilon_\mu^*,u,v,\overline u, \overline v,\dotsc\}$. In the fermionic case, writing the 
    Dirac field as
    \begin{equation}
     \Psi(x) = \int \frac{\dd^3k}{(2\pi)^3}  \frac{1}{\sqrt{2E_k}} 
     \left(a_k u \,e^{-ik\cdot x} + 
     b^\dagger_k v \,e^{ik\cdot x}\right)
    \end{equation}
    and taking the quantization condition $\{a_k,a^\dagger_{k'}\}=\{b_k,b^\dagger_{k'}\}=(2\pi)^3\delta^{(3)}(k-k')$, 
    one finds that $[a]=[b]=-3/2$ and 
    $[u]=[v]= [\alpha_{\text{fer}}]=1/2$. Similarly, one can also find that $[\alpha_{\text{bos}}] = 0$.
\end{itemize}
%
By taking into account the above points, we find that
\begin{equation}\label{eq:massM1}
    [\mathcal M_n] = \sum_{k=1}^m(4-[\Ocal_{j_k}]) + n_\partial + \mathsf D +\frac{1}{2}n_{\text{fer}}\,.
\end{equation}    
The mass dimension of $\mathcal M_n$ can be inferred also from the $S$-matrix element
    \begin{equation}
    \mel{0}{S}{n} = (2\pi)^4 \, \delta^{(4)}(p_1+\dotsb+p_n) \, i \mathcal M_n\,,
\end{equation}
where $[S]=0$ and $[\ket{n}] = [(\sqrt{2E} a^\dagger)^n \ket{0}] = -n$ with $n = n_{\text{fer}} + n_{\text{bos}}$, resulting in
\begin{equation}\label{eq:massM2}
    [\mathcal M_{n}] = 4 - n\,.
\end{equation}
From Eqs.~\eqref{eq:massM1} and \eqref{eq:massM2} 
and by taking into account that
    \begin{equation}
    [\mathcal O_i] = n_{\text{bos}} + \frac{3}{2}n_{\text{fer}} + n_\partial\,,
\end{equation}
it finally follows that
    \begin{equation}
    \mathsf D = 4 - [\mathcal O_i] + \sum_{k=1}^m([\Ocal_{j_k}]-4)\,. 
\end{equation}
